class Solution { public void buildMinHeap(int[] nums) { int n = nums.length; // Start from the last non-leaf node for (int i = (n / 2) - 1; i >= 0; i--) { heapifyDown(nums, i, n); } } private void heapifyDown(int[] nums, int i, int n) { while (true) { int smallest = i; int left = 2 * i + 1; int right = 2 * i + 2; // Check left child if (left < n && nums[left] < nums[smallest]) { smallest = left; } // Check right child if (right < n && nums[right] < nums[smallest]) { smallest = right; } // Already satisfies min-heap property if (smallest == i) { break; } // Swap int temp = nums[i]; nums[i] = nums[smallest]; nums[smallest] = temp; // Continue downward i = smallest; } } }
